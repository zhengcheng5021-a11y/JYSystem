    /* ================= 3. Interactive Falling Petal & Golden Dust Canvas ================= */
    const canvas = document.getElementById('zenCanvas');
    const ctx = canvas.getContext('2d');
    let width, height;
    let petals = [];
    let stardust = [];
    let touchRipples = [];

    function resizeCanvas() {
      const parent = canvas.parentElement;
      width = canvas.width = parent.clientWidth;
      height = canvas.height = parent.clientHeight;
    }
    window.addEventListener('resize', resizeCanvas);
    resizeCanvas();

    class Petal {
      constructor() {
        this.reset(true);
      }
      reset(init = false) {
        this.x = init ? Math.random() * width : Math.random() * width * 1.2 - width * 0.1;
        this.y = init ? Math.random() * height : -20;
        this.size = Math.random() * 7 + 6;
        this.speedY = Math.random() * 0.8 + 0.55;
        this.speedX = (Math.random() - 0.2) * 0.6 + 0.3;
        this.rotation = Math.random() * 360;
        this.rotationSpeed = (Math.random() - 0.5) * 1.6;
        this.flip = Math.random() * 360;
        this.flipSpeed = Math.random() * 2 + 1;
        this.alpha = Math.random() * 0.45 + 0.45;
        
        const palette = activeTheme.petalColors;
        this.color = palette[Math.floor(Math.random() * palette.length)];
      }
      update() {
        this.y += this.speedY;
        this.x += this.speedX + Math.sin(this.y * 0.015) * 0.5;
        this.rotation += this.rotationSpeed;
        this.flip += this.flipSpeed;

        if (this.y > height + 25 || this.x > width + 40 || this.x < -40) {
          this.reset();
        }
      }
      draw() {
        ctx.save();
        ctx.translate(this.x, this.y);
        ctx.rotate((this.rotation * Math.PI) / 180);
        ctx.scale(Math.cos((this.flip * Math.PI) / 180), 1);

        ctx.beginPath();
        ctx.moveTo(0, -this.size);
        ctx.bezierCurveTo(this.size * 0.9, -this.size * 0.9, this.size * 0.9, this.size * 0.5, 0, this.size);
        ctx.bezierCurveTo(-this.size * 0.9, this.size * 0.5, -this.size * 0.9, -this.size * 0.9, 0, -this.size);
        
        ctx.fillStyle = this.color;
        ctx.globalAlpha = this.alpha;
        ctx.shadowColor = activeTheme.rippleColor;
        ctx.shadowBlur = 4;
        ctx.fill();

        // Inner Petal Vein
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.4)';
        ctx.lineWidth = 0.7;
        ctx.beginPath();
        ctx.moveTo(0, -this.size * 0.7);
        ctx.lineTo(0, this.size * 0.6);
        ctx.stroke();

        ctx.restore();
      }
    }

    class Dust {
      constructor() {
        this.reset();
      }
      reset() {
        this.x = Math.random() * width;
        this.y = Math.random() * height;
        this.radius = Math.random() * 1.5 + 0.6;
        this.alpha = Math.random() * 0.7 + 0.2;
        this.alphaChange = (Math.random() - 0.5) * 0.015;
        this.speedY = -Math.random() * 0.3 - 0.1;
        this.speedX = (Math.random() - 0.5) * 0.3;
      }
      update() {
        this.y += this.speedY;
        this.x += this.speedX;
        this.alpha += this.alphaChange;
        if (this.alpha <= 0.15 || this.alpha >= 0.85) {
          this.alphaChange = -this.alphaChange;
        }
        if (this.y < -10) this.y = height + 10;
        if (this.x < -10) this.x = width + 10;
        if (this.x > width + 10) this.x = -10;
      }
      draw() {
        ctx.save();
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
        ctx.fillStyle = activeTheme.dustColor;
        ctx.shadowColor = '#FFFFFF';
        ctx.shadowBlur = 5;
        ctx.globalAlpha = Math.max(0, this.alpha);
        ctx.fill();
        ctx.restore();
      }
    }

    class Ripple {
      constructor(x, y) {
        this.x = x;
        this.y = y;
        this.radius = 2;
        this.maxRadius = 50;
        this.alpha = 0.6;
      }
      update() {
        this.radius += 1.6;
        this.alpha -= 0.018;
      }
      draw() {
        ctx.save();
        ctx.beginPath();
        ctx.arc(this.x, this.y, this.radius, 0, Math.PI * 2);
        ctx.strokeStyle = activeTheme.rippleColor;
        ctx.lineWidth = 1.2;
        ctx.globalAlpha = Math.max(0, this.alpha);
        ctx.stroke();
        ctx.restore();
      }
    }

    // Initialize particles
    for (let i = 0; i < 26; i++) petals.push(new Petal());
    for (let i = 0; i < 30; i++) stardust.push(new Dust());

    function renderLoop() {
      ctx.clearRect(0, 0, width, height);

      stardust.forEach(d => {
        d.update();
        d.draw();
      });

      petals.forEach(p => {
        p.update();
        p.draw();
      });

      for (let i = touchRipples.length - 1; i >= 0; i--) {
        const r = touchRipples[i];
        r.update();
        r.draw();
        if (r.alpha <= 0 || r.radius >= r.maxRadius) {
          touchRipples.splice(i, 1);
        }
      }

      requestAnimationFrame(renderLoop);
    }
    renderLoop();

    function addRipple(clientX, clientY) {
      const rect = canvas.getBoundingClientRect();
      const x = clientX - rect.left;
      const y = clientY - rect.top;
      if (x >= 0 && x <= width && y >= 0 && y <= height) {
        touchRipples.push(new Ripple(x, y));
      }
    }
    window.addEventListener('pointerdown', (e) => addRipple(e.clientX, e.clientY));
